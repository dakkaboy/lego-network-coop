clonewars_zig_dependencies(owner_sources "${CMAKE_CURRENT_SOURCE_DIR}/../../src/zig/owner_fixture_test.zig")
set_property(GLOBAL APPEND PROPERTY CLONEWARS_NATIVE_SOURCES ${owner_sources})
foreach(mode IN ITEMS Debug ReleaseSafe)
    set(fixture "${CMAKE_CURRENT_BINARY_DIR}/../native-zig/bin/owner-${mode}.dll")
    set_property(GLOBAL APPEND PROPERTY CLONEWARS_NATIVE_EXECUTABLES "${fixture}")
    foreach(target IN ITEMS A B)
        foreach(scenario RANGE 0 7)
            set(check "clonewars-owner-${mode}-${target}-${scenario}")
            add_test(NAME ${check} COMMAND "${CMAKE_CURRENT_BINARY_DIR}/../native-zig/bin/test_fixture_loader.exe" owner "${fixture}")
            set_tests_properties(${check} PROPERTIES TIMEOUT 20 LABELS "owner;native" ENVIRONMENT "OWNER_TARGET=${target};OWNER_SCENARIO=${scenario}")
        endforeach()
    endforeach()
endforeach()
