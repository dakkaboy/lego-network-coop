cmake_minimum_required(VERSION 3.25)
project(CloneWarsNetworkCoop LANGUAGES NONE)

if(NOT WIN32)
    message(FATAL_ERROR "The game targets x86 Windows.")
endif()
option(CLONEWARS_ZIG_CORE "Build synchronization modules with Zig" ON)
if(NOT CLONEWARS_ZIG_CORE)
    message(FATAL_ERROR "Zig is required.")
endif()
find_program(CLONEWARS_ZIG_EXECUTABLE NAMES zig REQUIRED)
execute_process(COMMAND "${CLONEWARS_ZIG_EXECUTABLE}" version OUTPUT_VARIABLE CLONEWARS_ZIG_VERSION OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT CLONEWARS_ZIG_VERSION STREQUAL "0.17.1")
    message(FATAL_ERROR "Zig 0.17.1 is required; set CLONEWARS_ZIG_EXECUTABLE.")
endif()

function(clonewars_zig_dependencies output)
    set(pending ${ARGN})
    set(result)
    while(pending)
        list(POP_FRONT pending current)
        get_filename_component(current "${current}" ABSOLUTE)
        if(current IN_LIST result)
            continue()
        endif()
        list(APPEND result "${current}")
        set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${current}")
        file(READ "${current}" source)
        string(REGEX MATCHALL [[@import\("[^"]+\.zig"\)]] imports "${source}")
        get_filename_component(directory "${current}" DIRECTORY)
        string(REGEX MATCHALL [[@embedFile\("[^"]+"\)]] embeds "${source}")
        foreach(embed IN LISTS embeds)
            string(REGEX REPLACE [[@embedFile\("([^"]+)"\)]] "\\1" relative "${embed}")
            list(APPEND result "${directory}/${relative}")
        endforeach()
        foreach(import IN LISTS imports)
            string(REGEX REPLACE [[@import\("([^"]+)"\)]] "\\1" relative "${import}")
            list(APPEND pending "${directory}/${relative}")
        endforeach()
    endwhile()
    set(${output} ${result} PARENT_SCOPE)
endfunction()

option(CLONEWARS_CONSTRUCTOR_QUERY_OBSERVER "Record native projectile construction queries" OFF)
option(CLONEWARS_HIT_ENTRY_OBSERVER "Record native hit entries" OFF)
option(CLONEWARS_TRACY "Enable Tracy profiling" OFF)
foreach(option IN ITEMS CONSTRUCTOR_QUERY_OBSERVER HIT_ENTRY_OBSERVER TRACY)
    if(CLONEWARS_${option})
        set(zig_${option} true)
    else()
        set(zig_${option} false)
    endif()
endforeach()
clonewars_zig_dependencies(dll_sources "${CMAKE_CURRENT_SOURCE_DIR}/src/zig/game_root.zig" "${CMAKE_CURRENT_SOURCE_DIR}/src/zig/proxy.zig")
set(game_dll "${CMAKE_CURRENT_BINARY_DIR}/clonewars-camera.dll")
set(proxy_dll "${CMAKE_CURRENT_BINARY_DIR}/d3d9.dll")
add_custom_command(OUTPUT "${game_dll}" "${proxy_dll}"
    COMMAND "${CLONEWARS_ZIG_EXECUTABLE}" build dll --prefix "${CMAKE_CURRENT_BINARY_DIR}" --prefix-exe-dir . -j4
        "-Dhit-entry=${zig_HIT_ENTRY_OBSERVER}" "-Dconstructor-query=${zig_CONSTRUCTOR_QUERY_OBSERVER}" "-Dtracy=${zig_TRACY}"
    WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
    DEPENDS ${dll_sources} build.zig build.zig.zon src/engine/proxy.def VERBATIM)
add_custom_target(clonewars-camera ALL DEPENDS "${game_dll}")
add_custom_target(d3d9 ALL DEPENDS "${proxy_dll}")
add_library(clonewars-camera-artifact SHARED IMPORTED GLOBAL)
set_target_properties(clonewars-camera-artifact PROPERTIES IMPORTED_LOCATION "${game_dll}")
add_library(d3d9-artifact SHARED IMPORTED GLOBAL)
set_target_properties(d3d9-artifact PROPERTIES IMPORTED_LOCATION "${proxy_dll}")

option(CLONEWARS_NETWORK_TESTS "Build checks that run without the game" OFF)
if(CLONEWARS_NETWORK_TESTS)
    clonewars_zig_dependencies(core_sources "${CMAKE_CURRENT_SOURCE_DIR}/src/zig/dll_root.zig")
    set(CLONEWARS_ZIG_LIBRARY "${CMAKE_CURRENT_BINARY_DIR}/clonewars-zig-core.lib")
    add_custom_command(OUTPUT "${CLONEWARS_ZIG_LIBRARY}"
        COMMAND "${CLONEWARS_ZIG_EXECUTABLE}" build-lib "${CMAKE_CURRENT_SOURCE_DIR}/src/zig/dll_root.zig"
            -target x86-windows-msvc -OReleaseSafe -lc -fcompiler-rt "-femit-bin=${CLONEWARS_ZIG_LIBRARY}"
        DEPENDS ${core_sources} VERBATIM)
    add_custom_target(clonewars-zig-build DEPENDS "${CLONEWARS_ZIG_LIBRARY}")
    enable_testing()
    add_subdirectory(tests)
endif()
